cmake_minimum_required(VERSION 3.24)
project(PhoneMirror VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
if(MSVC)
  add_compile_options(/utf-8)
endif()

add_library(pm_protocol protocol/src/protocol.cpp protocol/src/capture.cpp)
target_include_directories(pm_protocol PUBLIC protocol/include)
if(MSVC)
  target_compile_options(pm_protocol PRIVATE /W4 /utf-8)
else()
  target_compile_options(pm_protocol PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(pm-inspect windows/tools/inspect.cpp)
target_link_libraries(pm-inspect PRIVATE pm_protocol)

if(WIN32)
  add_library(pm_usb windows/transport/usb_client.cpp windows/transport/adb_session.cpp)
  target_compile_definitions(pm_usb PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(pm_usb PUBLIC windows/transport)
  target_link_libraries(pm_usb PUBLIC pm_protocol ws2_32)
  add_library(pm_playback windows/media/playback.cpp)
  target_include_directories(pm_playback PUBLIC windows/media)
  target_compile_definitions(pm_playback PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(pm_playback PUBLIC pm_protocol mfplat mf mfuuid wmcodecdspuuid d3d11 dxgi ole32)
  add_executable(pm-playback-check windows/tools/playback_check.cpp)
  target_link_libraries(pm-playback-check PRIVATE pm_playback)
  add_executable(pm-live-check windows/tools/live_check.cpp)
  target_link_libraries(pm-live-check PRIVATE pm_playback pm_usb)
  add_executable(PhoneMirror WIN32 windows/app/main.cpp)
  target_compile_definitions(PhoneMirror PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(PhoneMirror PRIVATE pm_playback pm_usb comdlg32 shell32)
  if(MSVC)
    target_compile_options(PhoneMirror PRIVATE /W4 /utf-8)
  endif()
  if(MINGW)
    target_link_options(PhoneMirror PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(pm-inspect PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(pm-playback-check PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(pm-live-check PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
endif()

if(BUILD_TESTING)
  add_executable(preferences-tests tests/preferences_tests.cpp)
  target_include_directories(preferences-tests PRIVATE windows/app)
  add_test(NAME preferences-recovery COMMAND preferences-tests)
  add_executable(experience-tests tests/experience_tests.cpp)
  target_include_directories(experience-tests PRIVATE windows/app)
  add_test(NAME product-experience COMMAND experience-tests)
  add_executable(gesture-tests tests/gesture_tests.cpp)
  target_link_libraries(gesture-tests PRIVATE pm_protocol)
  target_compile_definitions(gesture-tests PRIVATE PM_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/protocol/fixtures")
  add_test(NAME gesture-contract COMMAND gesture-tests)
  if(WIN32)
    add_test(NAME desktop-ui-flow COMMAND PhoneMirror --ui-smoke)
    set_tests_properties(desktop-ui-flow PROPERTIES TIMEOUT 10)
    foreach(scale IN ITEMS 96 144 192)
      add_test(NAME desktop-render-${scale} COMMAND PhoneMirror --snapshot "${CMAKE_CURRENT_BINARY_DIR}/ui-${scale}.bmp" ${scale})
      set_tests_properties(desktop-render-${scale} PROPERTIES TIMEOUT 10)
    endforeach()
    add_executable(adb-session-tests tests/adb_session_tests.cpp)
    target_link_libraries(adb-session-tests PRIVATE pm_usb)
    add_test(NAME desktop-connection COMMAND adb-session-tests)
    set_tests_properties(desktop-connection PROPERTIES TIMEOUT 20)
    add_executable(usb-tests tests/usb_tests.cpp)
    target_link_libraries(usb-tests PRIVATE pm_usb)
    add_test(NAME usb-transport COMMAND usb-tests)
    set_tests_properties(usb-transport PROPERTIES TIMEOUT 20)
  endif()
  add_executable(protocol-tests tests/protocol_tests.cpp)
  target_link_libraries(protocol-tests PRIVATE pm_protocol)
  target_compile_definitions(protocol-tests PRIVATE PM_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/protocol/fixtures")
  add_test(NAME protocol-contract COMMAND protocol-tests)
  add_executable(capture-tests tests/capture_tests.cpp)
  target_link_libraries(capture-tests PRIVATE pm_protocol)
  add_test(NAME capture-semantics COMMAND capture-tests)
endif()
